\section*{PART I: Multiple Choice (10 points)}
\textit{Circle the letter of the best answer. 2 points each.}

\vspace{10pt}

\textbf{1.} Which of the following is NOT a proposition?

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item The Earth is flat.
    \item $3 + 4 = 8$
    \item $x^2 = 4$
    \item Every even number greater than 2 is the sum of two primes.
\end{enumerate}

\vspace{25pt}

\textbf{2.} The logical equivalence $p \to q \equiv \neg p \lor q$ is called:

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item De Morgan's Law
    \item Implication Law
    \item Double Negation
    \item Contrapositive Law
\end{enumerate}

\vspace{25pt}

\textbf{3.} What is the truth value of $\forall x\, (x^2 > x)$ if the domain is all positive real numbers greater than 1?

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item True
    \item False
\end{enumerate}

\vspace{25pt}

\textbf{4.} The negation of $\forall x \exists y\, P(x, y)$ is:

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item $\exists x \forall y\, P(x, y)$
    \item $\forall x \exists y\, \neg P(x, y)$
    \item $\exists x \forall y\, \neg P(x, y)$
    \item $\exists x \exists y\, \neg P(x, y)$
\end{enumerate}

\vspace{25pt}

\textbf{5.} The argument form $p \to q$, $q \to r$, therefore $p \to r$ is called:

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item Modus Ponens
    \item Modus Tollens
    \item Hypothetical Syllogism
    \item Resolution
\end{enumerate}

\begin{enumerate}[label=(\roman*)]

\section*{PART III: Problem Solving (25 points)}

\textbf{Question 5.} (6 points) Identify the rule of inference used in each argument.

\begin{enumerate}[label=(\roman*)]
    \item If $n$ is an even integer, then $n^2$ is even. The integer 6 is even. Therefore, $36$ is even.
    
    \vspace{40pt}
    
    \item Jane will take CS 101 or Math 201. Jane will not take Math 201. Therefore, Jane will take CS 101.
    
    \vspace{40pt}
    
    \item If today is Friday, we have pizza. We do not have pizza. Therefore, today is not Friday.
    
    \vspace{40pt}
\end{enumerate}

\textbf{Question 6.} (7 points) Prove that $\neg(p \lor q) \lor (\neg p \land q) \equiv \neg p$ without using truth tables. List each logical law applied in your proof.

\vspace{120pt}

$$\forall x \forall y\, (P(x) \land Q(y) \to R(x, y))$$

\vspace{80pt}

\textbf{Question 8.} (8 points) Using rules of inference, prove that the following argument is valid. Clearly show each step and identify the rule used.

\textbf{Premises:}
