\section*{PART I: Multiple Choice (10 points)}
\textit{Circle the letter of the best answer. 2 points each.}

\vspace{10pt}

\textbf{1.} The statement ``If $p$ then $q$'' is logically equivalent to:

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item $p \land q$
    \item $\neg p \lor q$
    \item $q \to p$
    \item $\neg q \lor \neg p$
\end{enumerate}

\vspace{25pt}

\textbf{2.} Which statement represents ``Some students passed the exam''?

Let $S(x)$ = ``$x$ is a student'' and $P(x)$ = ``$x$ passed the exam''.

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item $\forall x\, (S(x) \to P(x))$
    \item $\forall x\, (S(x) \land P(x))$
    \item $\exists x\, (S(x) \to P(x))$
    \item $\exists x\, (S(x) \land P(x))$
\end{enumerate}

\vspace{25pt}

\textbf{3.} The inverse of $p \to q$ is:

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item $\neg q \to \neg p$
    \item $q \to p$
    \item $\neg p \to \neg q$
    \item $\neg(p \to q)$
\end{enumerate}

\vspace{25pt}

\textbf{4.} A compound proposition that is always false is called a:

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item Tautology
    \item Contingency
    \item Contradiction
    \item Fallacy
\end{enumerate}

\vspace{25pt}

\textbf{5.} The rule $\dfrac{p \land q}{\therefore p}$ is called:

\begin{enumerate}[label=(\Alph*),noitemsep]
    \item Conjunction
    \item Simplification
    \item Addition
    \item Absorption
\end{enumerate}
